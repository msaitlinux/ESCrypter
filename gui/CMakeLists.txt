cmake_minimum_required(VERSION 3.16)
project(escrypter_gui VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Svg Concurrent)
find_package(CURL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SODIUM REQUIRED IMPORTED_TARGET libsodium)
qt_add_executable(escrypter-gui main.cpp window.cpp window.h jobs.cpp jobs.h
  remote.cpp remote.h transfers.cpp transfers.h resources.qrc)
target_link_libraries(escrypter-gui PRIVATE Qt6::Widgets Qt6::Svg Qt6::Concurrent CURL::libcurl PkgConfig::SODIUM)
install(TARGETS escrypter-gui DESTINATION bin)
option(ESCRYPTER_TESTS "Build GUI regression checks" OFF)
if(ESCRYPTER_TESTS)
  qt_add_executable(escrypter-gui-tests ../tests/gui_smoke.cpp window.cpp window.h jobs.cpp jobs.h remote.cpp remote.h transfers.cpp transfers.h resources.qrc)
  target_include_directories(escrypter-gui-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_libraries(escrypter-gui-tests PRIVATE Qt6::Widgets Qt6::Svg Qt6::Concurrent CURL::libcurl PkgConfig::SODIUM)
endif()
